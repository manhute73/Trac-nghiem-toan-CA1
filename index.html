// Thuật toán cắt khung vừa vặn 100% không lo trồi sụt function cropBoundingBox(sourceCanvas, startY, endY) { const width = sourceCanvas.width; const height = Math.max(50, endY - startY); // Tạo canvas tạm thời để chứa câu hỏi const tempCanvas = document.createElement('canvas'); tempCanvas.width = width; tempCanvas.height = height; const ctx = tempCanvas.getContext('2d'); // Vẽ đúng vùng từ startY đến endY ctx.drawImage(sourceCanvas, 0, startY, width, height, 0, 0, width, height); // Xén lề thừa (Trim whitespace) để đề luôn căn giữa khung const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; let top = height, bottom = 0; for (let y = 0; y < height; y++) { for (let x = 0; x < width; x++) { const alpha = data[(y * width + x) * 4 + 3]; const isNotWhite = data[(y * width + x) * 4] < 240; // Phát hiện pixel chữ/ma trận if (alpha > 0 && isNotWhite) { if (y < top) top = y; if (y > bottom) bottom = y; } } } // Nếu tìm thấy vùng chứa chữ thực tế, cắt sát lề thêm 10px padding const realHeight = Math.max(40, (bottom - top) + 20); const finalCanvas = document.createElement('canvas'); finalCanvas.width = width; finalCanvas.height = realHeight; finalCanvas.getContext('2d').drawImage( tempCanvas, 0, Math.max(0, top - 10), width, realHeight, 0, 0, width, realHeight ); return finalCanvas.toDataURL('image/jpeg', 0.95); }
